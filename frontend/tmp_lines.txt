  18:     const date = new Date(alerte.reported_at).toLocaleDateString('fr-FR', {
  19:         day: '2-digit', month: 'long', year: 'numeric',

  42:                     </svg>

  46:                     </span>

  74: // ── Panneau principal ─────────────────────────────────────────────────────────
  75: export default function PanneauDetailZone({ zone, onClose }) {
  76:     const [incidents, setIncidents] = useState([]);
  77:     const [loading, setLoading]     = useState(true);
  78: 
  79:     const cfg = TYPE_CONFIG[zone.zone_type] ?? TYPE_CONFIG.accident_prone;
  80:     const dist = zone.periods_distribution ?? {};
  81: 
  82:     useEffect(() => {
  83:         setLoading(true);
  84:         api.get(`/zones-a-risque/${zone.id}/alertes`)
  85:             .then(({ data }) => {
  86:                 const list = Array.isArray(data) ? data : (data?.data ?? data?.alertes ?? []);
  87:                 setIncidents(list);
  88:             })
  89:             .catch(() => {})
  90:             .finally(() => setLoading(false));
  91:     }, [zone.id]);
  92: 
  93:     return (
  94:         <div className="flex flex-col h-full min-h-0 overflow-hidden bg-white">
  95:             <div className="flex-1 min-h-0 overflow-y-auto">
  96:                 <div>
  97:                     {/* Header */}
  98:                     <div className={`${cfg.bg} border-b border-slate-200 px-5 py-4`}>
  99:                 <div className="flex items-start justify-between gap-3">
 100:                     <div className="flex items-start gap-3">
 101:                         <div className={`w-10 h-10 rounded-xl ${cfg.iconBg} ${cfg.iconColor} flex items-center justify-center shrink-0`}>
 102:                             {cfg.icon}
 103:                         </div>
 104:                         <div>
 105:                             <span className={`inline-flex px-2 py-0.5 rounded-full text-xs font-semibold ${cfg.badge}`}>
 106:                                 {cfg.label}
 107:                             </span>
 108:                             <h2 className="font-bold text-slate-800 text-base mt-1">
 109:                                 {zone.city}
 110:                                 {zone.district && (
 111:                                     <span className="font-normal text-slate-500 text-sm"> — {zone.district}</span>
 112:                                 )}
 113:                             </h2>
 114:                         </div>
 115:                     </div>
 116:                     <button
 117:                         onClick={onClose}
 118:                         className="w-8 h-8 flex items-center justify-center rounded-full hover:bg-black/10 text-slate-500 transition-colors shrink-0"
 119:                     >
 120:                         <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2.5}>
 121:                             <path strokeLinecap="round" strokeLinejoin="round" d="M6 18 18 6M6 6l12 12" />
 122:                         </svg>
 123:                     </button>
 124:                 </div>
 125:             </div>
 126: 
 127:                 {/* Stats synthèse */}
 128:                 <div className="mt-4 space-y-2">
 129:                     <div className="bg-white rounded-lg px-3 py-2 border border-slate-200 text-center">
 130:                         <p className="text-xl font-bold text-slate-800">
 131:                             {loading ? zone.incident_count : incidents.length}
 132:                         </p>
 133:                         <p className="text-xs text-slate-500 mt-0.5">Signalements</p>
 134:                     </div>
 135:                     {Object.keys(dist).length > 0 && (
 136:                         <div className="bg-white rounded-lg px-4 py-3 border border-slate-200 space-y-2">
 137:                             <p className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Répartition des accidents</p>
 138:                             {Object.entries(PERIODS_CONFIG).map(([key, { label, hours }]) => (
 139:                                 <div key={key} className="flex items-center justify-between text-sm">
 140:                                     <span className="text-slate-600">{label} <span className="text-xs text-slate-400">({hours})</span></span>
 141:                                     <span className="font-semibold text-slate-800">{dist[key] ?? 0}</span>
 142:                                 </div>
 143:                             ))}
 144:                         </div>
 145:                     )}
 146:                 </div>
 147:             </div>
 148: 
 149:             {/* Corps — liste des incidents */}
 150:             <div className="flex-1 overflow-y-auto px-5 py-4">
 151:                 <div className="flex items-center justify-between mb-1">
 152:                     <h3 className="text-sm font-semibold text-slate-700">Incidents liés</h3>
 153:                     {!loading && (
 154:                         <span className="text-xs text-slate-400">
 155:                             {incidents.length} signalement{incidents.length !== 1 ? 's' : ''}
 156:                         </span>
 157:                     )}
 158:                 </div>
 159:                 <p className="text-xs text-slate-400 mb-4">
 160:                     Incidents ayant généré cette zone
 161:                 </p>
 162: 
 163:                 {loading && (
 164:                     <div className="py-8 flex flex-col items-center gap-2">
 165:                         <div className="w-6 h-6 border-2 border-blue-500 border-t-transparent rounded-full animate-spin" />
 166:                         <p className="text-xs text-slate-400">Chargement…</p>
 167:                     </div>
 168:                 )}
 169: 
 170:                 {!loading && incidents.length === 0 && (
 171:                     <div className="py-10 text-center bg-slate-50 rounded-xl border border-slate-200">
 172:                         <svg className="w-8 h-8 text-slate-300 mx-auto mb-2" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
 173:                             <path strokeLinecap="round" strokeLinejoin="round" d="M9 12h3.75M9 15h3.75M9 18h3.75m3 .75H18a2.25 2.25 0 0 0 2.25-2.25V6.108c0-1.135-.845-2.098-1.976-2.192a48.424 48.424 0 0 0-1.123-.08m-5.801 0c-.065.21-.1.433-.1.664 0 .414.336.75.75.75h4.5a.75.75 0 0 0 .75-.75 2.25 2.25 0 0 0-.1-.664m-5.8 0A2.251 2.251 0 0 1 13.5 2.25H15c1.012 0 1.867.668 2.15 1.586m-5.8 0c-.376.023-.75.05-1.124.08C9.095 4.01 8.25 4.973 8.25 6.108V8.25m0 0H4.875c-.621 0-1.125.504-1.125 1.125v11.25c0 .621.504 1.125 1.125 1.125h9.75c.621 0 1.125-.504 1.125-1.125V9.375c0-.621-.504-1.125-1.125-1.125H8.25Z" />
 174:                         </svg>
 175:                         <p className="text-sm text-slate-500 font-medium">Aucun incident trouvé</p>
 176:                         <p className="text-xs text-slate-400 mt-1">
 177:                             Aucun incident de ce type n'a été signalé dans cette ville.
 178:                         </p>
 179:                     </div>
 180:                 )}
 181: 
 182:                 {!loading && incidents.length > 0 && (
 183:                     <div>
 184:                         {incidents.map((alerte) => (
 185:                             <IncidentItem key={alerte.id} alerte={alerte} />
 186:                         ))}
 187:                     </div>
 188:                 )}
 189:             </div>
 190:         </div>
 191:     );
